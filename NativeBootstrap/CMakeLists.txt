cmake_minimum_required(VERSION 3.21)
project(BSIPADoorstop LANGUAGES C)
if(NOT MSVC)
  message(FATAL_ERROR "The BSIPA bootstrap requires the Visual C++ Windows toolchain.")
endif()
if(NOT EXISTS "${DOORSTOP_SOURCE_DIR}/src/windows/entrypoint.c")
  message(FATAL_ERROR "Provide the pinned Doorstop source directory.")
endif()

# Retain upstream's export list and proxy generation, without requiring XMake.
file(STRINGS "${DOORSTOP_SOURCE_DIR}/src/windows/proxy/proxylist.txt" exports)
set(PROXY_DEFS "")
set(PROXY_ADDS "")
set(PROXY_FUNCS "")
set(EXPORT_FUNCS "")
foreach(name IN LISTS exports)
  string(STRIP "${name}" name)
  if(name STREQUAL "")
    continue()
  endif()
  string(APPEND PROXY_DEFS "static FARPROC __${name}__;\n")
  string(APPEND PROXY_ADDS "__${name}__ = GetProcAddress((HMODULE)dll, \"${name}\");\n")
  string(APPEND PROXY_FUNCS "void *exp_${name}() { return __${name}__(); }\n")
  string(APPEND EXPORT_FUNCS "${name} = exp_${name}\n")
endforeach()
configure_file("${DOORSTOP_SOURCE_DIR}/src/windows/build_tools/proxy.c.in" proxy.c)
configure_file("${DOORSTOP_SOURCE_DIR}/src/windows/build_tools/dll.def.in" dll.def)

file(READ "${DOORSTOP_SOURCE_DIR}/src/windows/entrypoint.c" entry)
set(anchor "    inject(paths);")
string(FIND "${entry}" "${anchor}" found)
if(found EQUAL -1)
  message(FATAL_ERROR "Pinned Doorstop bootstrap anchor changed.")
endif()
string(REPLACE "${anchor}" "    bsipa_configure_fpfc(paths, &new_cmdline_args, &new_cmdline_args_narrow);\n${anchor}" entry "${entry}")
string(REPLACE "#include \"entrypoint.h\"" "#include \"entrypoint.h\"\n#include \"fpfc.h\"" entry "${entry}")
string(REPLACE "#define HOOK_SYS(mod, from, to) ok &= iat_hook(mod, \"kernel32.dll\", &from, &to)"
  "#define HOOK_SYS(mod, from, to) do { bool_t hooked = iat_hook(mod, \"kernel32.dll\", &from, &to); ok &= hooked; bsipa_record_hook(#from, hooked); } while (0)" entry "${entry}")
string(REPLACE "    paths_free(paths);" "    bsipa_finish_hooks();\n    paths_free(paths);" entry "${entry}")
string(REPLACE "        SetEnvironmentVariableW(L\"DOORSTOP_DISABLE\", NULL);" "    {\n        SetEnvironmentVariableW(L\"DOORSTOP_DISABLE\", NULL);\n        bsipa_cleanup_fpfc();\n    }" entry "${entry}")
# The upstream ANSI override overwrites the caller's filename buffer. Keep the
# replacement in its own allocation, as the wide-character hook already does.
set(ansi_old "        char *narrowed_boot_config_override =\n            narrow(config.boot_config_override);\n        memcpy(actual_file_name, narrowed_boot_config_override,\n               strlen(config.boot_config_override));\n        free(narrowed_boot_config_override);")
string(REPLACE "${ansi_old}" "        actual_file_name = narrow(config.boot_config_override);" entry "${entry}")
set(ansi_return "    return CreateFileA(actual_file_name, dwDesiredAccess, dwShareMode,\n                       lpSecurityAttributes, dwCreationDisposition,\n                       dwFlagsAndAttributes, hTemplateFile);")
string(REPLACE "${ansi_return}" "    HANDLE result = CreateFileA(actual_file_name, dwDesiredAccess, dwShareMode,\n                       lpSecurityAttributes, dwCreationDisposition,\n                       dwFlagsAndAttributes, hTemplateFile);\n    if (actual_file_name != lpFileName) free(actual_file_name);\n    return result;" entry "${entry}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/entrypoint.c" "${entry}")
file(GLOB sources "${DOORSTOP_SOURCE_DIR}/src/*.c" "${DOORSTOP_SOURCE_DIR}/src/config/*.c"
  "${DOORSTOP_SOURCE_DIR}/src/util/*.c" "${DOORSTOP_SOURCE_DIR}/src/runtimes/*.c"
  "${DOORSTOP_SOURCE_DIR}/src/windows/*.c")
list(FILTER sources EXCLUDE REGEX "/windows/entrypoint\\.c$")
add_library(winhttp SHARED ${sources} "${CMAKE_CURRENT_BINARY_DIR}/entrypoint.c"
  "${CMAKE_CURRENT_BINARY_DIR}/proxy.c" "${CMAKE_CURRENT_BINARY_DIR}/dll.def" fpfc.c)
target_include_directories(winhttp PRIVATE "${DOORSTOP_SOURCE_DIR}/src/windows" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(winhttp PRIVATE UNICODE _UNICODE)
target_compile_options(winhttp PRIVATE /GS- /Ob2 /Os /GL- /FS)
target_link_options(winhttp PRIVATE /NODEFAULTLIB /ENTRY:DllEntry /DYNAMICBASE:NO)
target_link_libraries(winhttp PRIVATE kernel32 shell32 user32)
set_property(TARGET winhttp PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
